// Invoice Calculation Logic
const calculateInvoice = (items, taxRate) => {
  const subtotal = items.reduce((sum, item) => {
    return sum + (item.quantity * item.rate);
  }, 0);
  const tax = subtotal * (taxRate / 100);
  return { subtotal, tax, total: subtotal + tax };
};

const authMiddleware = async (req, res, next) => {
  try {
    const token = req.header('Authorization');
    const decoded = jwt.verify(token, SECRET);
    req.user = await User.findById(decoded.userId);
    next();
  } catch (error) {
    res.status(401).json({ error: 'Unauthorized' });
  }
};

Chapter 5

Challenges Faced

Problems & Solutions

Timer Persistence

Timer state was lost on page navigation. Solved using localStorage to persist timer data across components.

PDF Generation

Server-side PDF generation was complex. Implemented client-side PDF using jsPDF library for instant downloads.

Data Isolation

Ensuring users only see their own data. Every database query includes user_id filter for strict isolation.

CORS Issues

Cross-origin requests blocked in production. Configured proper CORS headers on the backend server.
